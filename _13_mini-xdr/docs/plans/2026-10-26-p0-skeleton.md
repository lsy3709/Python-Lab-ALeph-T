```python
"""공개 저장소에 비밀(API 키·토큰·개인키)이 섞여 들어가지 않았는지 검사한다.

사용: python -m app.secret_scan [경로]   → 발견 시 종료 코드 1 (값은 출력하지 않음)
"""
from __future__ import annotations

import re
import sys
from collections.abc import Iterator
from dataclasses import dataclass
from pathlib import Path

# 실제 키 형식만 잡도록 길이 하한을 둔다. 자리표시자(<YOUR_API_KEY>)는 통과.
PATTERNS: dict[str, re.Pattern[str]] = {
    "anthropic_api_key": re.compile(r"sk-ant-[A-Za-z0-9_\-]{20,}"),
    "google_api_key": re.compile(r"AIza[0-9A-Za-z_\-]{35}"),
    "aws_access_key_id": re.compile(r"AKIA[0-9A-Z]{16}"),
    "private_key_block": re.compile(r"-----BEGIN (?:RSA |EC |OPENSSH |DSA )?PRIVATE KEY-----"),
    "github_token": re.compile(r"gh[pousr]_[A-Za-z0-9]{36,}"),
    "slack_token": re.compile(r"xox[baprs]-[A-Za-z0-9-]{10,}"),
    "notion_token": re.compile(r"(?:secret|ntn)_[A-Za-z0-9]{40,}"),
}

SKIP_DIRS = {".git", ".venv", "__pycache__", ".pytest_cache", "node_modules"}
TEXT_SUFFIXES = {
    "", ".py", ".md", ".txt", ".sql", ".ini", ".toml", ".cfg",
    ".yml", ".yaml", ".json", ".example", ".sh", ".ps1",
}


@dataclass(frozen=True)
class Finding:
    path: Path
    line: int
    kind: str


def scan_text(text: str) -> list[tuple[int, str]]:
    """문자열에서 (줄 번호, 종류) 목록을 돌려준다."""
    hits: list[tuple[int, str]] = []
    for lineno, line in enumerate(text.splitlines(), start=1):
        for kind, pattern in PATTERNS.items():
            if pattern.search(line):
                hits.append((lineno, kind))
    return hits


def _is_secret_env_file(name: str) -> bool:
    """.gitignore 로 커밋에서 빠지는 실제 비밀 파일(.env, .env.local …)은 검사 대상이 아니다."""
    return name == ".env" or (name.startswith(".env.") and name != ".env.example")


def iter_files(root: Path) -> Iterator[Path]:
    for path in sorted(root.rglob("*")):
        if not path.is_file():
            continue
        if any(part in SKIP_DIRS for part in path.relative_to(root).parts):
            continue
        if _is_secret_env_file(path.name) or path.suffix not in TEXT_SUFFIXES:
            continue
        yield path


def scan_path(root: Path) -> list[Finding]:
    findings: list[Finding] = []
    for path in iter_files(root):
        text = path.read_text(encoding="utf-8", errors="ignore")
        findings.extend(Finding(path, lineno, kind) for lineno, kind in scan_text(text))
    return findings


def main(argv: list[str] | None = None) -> int:
    args = sys.argv[1:] if argv is None else argv
    root = Path(args[0]) if args else Path(__file__).resolve().parent.parent
    findings = scan_path(root)
    for f in findings:
        print(f"[비밀 의심] {f.path}:{f.line} ({f.kind})")
    if not findings:
        print(f"비밀 스캔 통과: {root}")
    return 1 if findings else 0


if __name__ == "__main__":
    raise SystemExit(main())
```

```python
"""mini-xdr 설정 — 환경변수(.env)에서 읽는다.

API 키가 없거나 자리표시자(<YOUR_API_KEY>)면 LLM 을 끄고 규칙 판정으로 동작한다
(키 없는 학생도 수업 참여 가능 — [186] 설계 원칙 4).
"""
from __future__ import annotations

from typing import Literal

from pydantic import Field, SecretStr
from pydantic_settings import BaseSettings, SettingsConfigDict


def _has(secret: SecretStr | None) -> bool:
    if secret is None:
        return False
    value = secret.get_secret_value().strip()
    return bool(value) and not value.startswith("<")


class Settings(BaseSettings):
    model_config = SettingsConfigDict(env_file=".env", env_file_encoding="utf-8", extra="ignore")

    # LLM
    llm_provider: Literal["claude", "gemini", "none"] = "claude"
    anthropic_api_key: SecretStr | None = None
    claude_model: str = "claude-haiku-5-5"
    claude_escalate_model: str = "claude-sonnet-5-5"
    gemini_api_key: SecretStr | None = None
    gemini_model: str | None = None
    max_daily_llm_calls: int = Field(200, ge=0)
    prompt_cache_ttl: Literal["5m", "1h"] = "5m"

    # 수집·상관
    graylog_url: str = "http://localhost:9000"
    graylog_token: SecretStr | None = None
    poll_interval_sec: int = Field(30, ge=5)
    correlation_window_min: int = Field(10, ge=1)

    # 저장소
    mysql_host: str = "localhost"
    mysql_port: int = 3306
    mysql_user: str = "xdr"
    mysql_password: SecretStr | None = None
    mysql_db: str = "mini_xdr"

    # 대응
    n8n_webhook_url: str | None = None

    @property
    def llm_disabled_reason(self) -> str | None:
        """LLM 을 쓸 수 없는 이유. 쓸 수 있으면 None."""
        if self.llm_provider == "none":
            return "LLM_PROVIDER=none"
        if self.llm_provider == "claude" and not _has(self.anthropic_api_key):
            return "ANTHROPIC_API_KEY 없음"
        if self.llm_provider == "gemini":
            if not _has(self.gemini_api_key):
                return "GEMINI_API_KEY 없음"
            if not self.gemini_model:
                return "GEMINI_MODEL 없음"
        return None

    @property
    def llm_enabled(self) -> bool:
        return self.llm_disabled_reason is None
```

```python
"""스키마를 일회용 MySQL 8.0 컨테이너에 실제로 적용해 본다(도커 필요).

실행: python -m pytest -m integration -v
"""
import shutil
import subprocess
import time
import uuid

import pytest

from app.schema import SCHEMA_PATH, TABLES

pytestmark = pytest.mark.integration

ROOT_PW = "xdr_throwaway_test_pw"  # 테스트가 끝나면 삭제되는 일회용 컨테이너 전용


def _docker_ready() -> bool:
    if shutil.which("docker") is None:
        return False
    return subprocess.run(["docker", "info"], capture_output=True).returncode == 0


def _mysql(name: str, *args: str, input_sql: str | None = None) -> subprocess.CompletedProcess:
    # -h 127.0.0.1 TCP 접속: 초기화용 임시 서버(소켓 전용)가 아닌 최종 서버에만 붙는다.
    cmd = ["docker", "exec", "-i", name, "mysql", "-h", "127.0.0.1", "--protocol=TCP",
           "-uroot", f"-p{ROOT_PW}", *args]
    return subprocess.run(cmd, input=input_sql, capture_output=True, text=True, encoding="utf-8")


@pytest.fixture(scope="module")
def mysql_container():
    if not _docker_ready():
        pytest.skip("docker 를 쓸 수 없음")
    name = f"mini-xdr-schema-test-{uuid.uuid4().hex[:8]}"
    subprocess.run(["docker", "run", "-d", "--rm", "--name", name,
                    "-e", f"MYSQL_ROOT_PASSWORD={ROOT_PW}", "mysql:8.0"],
                   check=True, capture_output=True)
    try:
        deadline = time.time() + 180
        while _mysql(name, "-e", "SELECT 1").returncode != 0:
            if time.time() > deadline:
                pytest.fail("MySQL 준비 시간 초과")
            time.sleep(2)
        yield name
    finally:
        subprocess.run(["docker", "rm", "-f", name], capture_output=True)


@pytest.fixture(scope="module")
def schema_applied(mysql_container):
    sql = SCHEMA_PATH.read_text(encoding="utf-8")
    for attempt in (1, 2):  # 두 번 적용해도 오류가 없어야 한다(멱등)
        r = _mysql(mysql_container, input_sql=sql)
        assert r.returncode == 0, f"{attempt}회차 적용 실패: {r.stderr}"
    return mysql_container


def test_all_tables_exist(schema_applied):
    r = _mysql(schema_applied, "-N", "-e", "SHOW TABLES FROM mini_xdr")
    assert set(r.stdout.split()) == set(TABLES)


def test_raw_ref_unique_blocks_duplicate_ingest(schema_applied):
    ins = ("INSERT INTO mini_xdr.events_norm (ts, layer, source, event_type, raw_ref) "
           "VALUES (NOW(3), 'app', 'board', 'web-scan', 'gl-dup-1');")
    assert _mysql(schema_applied, "-e", ins).returncode == 0
    dup = _mysql(schema_applied, "-e", ins)
    assert dup.returncode != 0 and "Duplicate entry" in dup.stderr


def test_incident_delete_cascades_to_links(schema_applied):
    sql = (
        "USE mini_xdr;"
        "INSERT INTO events_norm (ts, layer, source, event_type, raw_ref) VALUES (NOW(3),'host','wazuh','rule','gl-casc-1');"
        "INSERT INTO incidents (entity_key, opened_at, last_event_at, layers) VALUES ('ip:test', NOW(3), NOW(3), JSON_ARRAY('host'));"
        "INSERT INTO incident_events VALUES (LAST_INSERT_ID(), (SELECT id FROM events_norm WHERE raw_ref='gl-casc-1'));"
        "DELETE FROM incidents WHERE entity_key='ip:test';"
        "SELECT COUNT(*) FROM incident_events;"
    )
    r = _mysql(schema_applied, "-N", "-e", sql)
    assert r.returncode == 0, r.stderr
    assert r.stdout.split()[-1] == "0"
```

```sql
-- mini-xdr 저장소 스키마 (P0) — MySQL 8.0
-- 적용(PowerShell): Get-Content -Raw sql\001_schema.sql | docker exec -i <MySQL컨테이너> mysql -uroot -p<루트비번>
-- 여러 번 적용해도 안전하다(IF NOT EXISTS). 게시판 DB 의 incidents(SOAR 티켓)와 섞이지 않도록 전용 DB 를 쓴다.

CREATE DATABASE IF NOT EXISTS mini_xdr CHARACTER SET utf8mb4 COLLATE utf8mb4_0900_ai_ci;
USE mini_xdr;

-- 1) 정규화 이벤트 — 출처가 달라도 같은 모양으로
CREATE TABLE IF NOT EXISTS events_norm (
  id          BIGINT UNSIGNED AUTO_INCREMENT PRIMARY KEY,
  ts          DATETIME(3)      NOT NULL,
  layer       ENUM('app','host','network','data','behavior') NOT NULL,
  source      VARCHAR(32)      NOT NULL,            -- board, wazuh, pfsense, mysql_audit, anomaly
  src_ip      VARCHAR(45)      NULL,                -- IPv6 대비 45자
  user_name   VARCHAR(128)     NULL,
  asset       VARCHAR(128)     NULL,
  event_type  VARCHAR(64)      NOT NULL,            -- login-bruteforce, web-scan, rule ...
  rule_id     VARCHAR(16)      NULL,
  level       TINYINT UNSIGNED NULL,                -- Wazuh level 0~15
  mitre       VARCHAR(64)      NULL,                -- T1110 등(여러 개면 쉼표)
  raw_ref     VARCHAR(64)      NOT NULL,            -- Graylog 메시지 ID(원본 참조)
  ingested_at DATETIME(3)      NOT NULL DEFAULT CURRENT_TIMESTAMP(3),
  UNIQUE KEY uq_events_raw_ref (raw_ref),           -- 같은 원본 중복 적재 방지
  KEY ix_events_ip_ts (src_ip, ts),
  KEY ix_events_user_ts (user_name, ts)
) ENGINE=InnoDB;

-- 2) 인시던트 — 관련 이벤트 묶음
CREATE TABLE IF NOT EXISTS incidents (
  id               BIGINT UNSIGNED AUTO_INCREMENT PRIMARY KEY,
  entity_key       VARCHAR(160) NOT NULL,           -- ip:203.0.113.7 또는 user:alice
  status           ENUM('open','closed') NOT NULL DEFAULT 'open',
  opened_at        DATETIME(3)  NOT NULL,
  last_event_at    DATETIME(3)  NOT NULL,
  layers           JSON         NOT NULL,           -- ["app","host"]
  event_count      INT UNSIGNED NOT NULL DEFAULT 0,
  max_level        TINYINT UNSIGNED NULL,
  fingerprint      CHAR(64)     NULL,               -- SHA-256 hex (호출 깔때기 G3)
  rule_floor       ENUM('P1','P2','P3','P4') NULL,  -- AI 가 이보다 낮출 수 없는 하한
  current_priority ENUM('P1','P2','P3','P4') NULL,
  last_triaged_at  DATETIME(3)  NULL,
  material_state   JSON         NULL,               -- 디바운스(G4) 비교용: 마지막 판정 시점 상태
  KEY ix_incidents_entity_status (entity_key, status),
  KEY ix_incidents_fingerprint (fingerprint),
  KEY ix_incidents_priority (status, current_priority)
) ENGINE=InnoDB;

-- 3) 인시던트와 이벤트 연결
CREATE TABLE IF NOT EXISTS incident_events (
  incident_id BIGINT UNSIGNED NOT NULL,
  event_id    BIGINT UNSIGNED NOT NULL,
  PRIMARY KEY (incident_id, event_id),
  KEY ix_ie_event (event_id),
  CONSTRAINT fk_ie_incident FOREIGN KEY (incident_id) REFERENCES incidents(id)   ON DELETE CASCADE,
  CONSTRAINT fk_ie_event    FOREIGN KEY (event_id)    REFERENCES events_norm(id) ON DELETE CASCADE
) ENGINE=InnoDB;

-- 4) 행위 기준선 — 엔티티별 특징의 평균·표준편차
CREATE TABLE IF NOT EXISTS baselines (
  entity_type ENUM('user','ip') NOT NULL,
  entity      VARCHAR(128) NOT NULL,
  feature     VARCHAR(32)  NOT NULL,               -- login_hour, req_per_min, ratio_404, fail_ratio, path_diversity
  mean        DOUBLE       NOT NULL,
  std         DOUBLE       NOT NULL,
  n           INT UNSIGNED NOT NULL,               -- 표본 수(30 미만이면 기준선 부족)
  updated_at  DATETIME(3)  NOT NULL,
  PRIMARY KEY (entity_type, entity, feature)
) ENGINE=InnoDB;

-- 5) 이상 점수 이력
CREATE TABLE IF NOT EXISTS anomaly_scores (
  id          BIGINT UNSIGNED AUTO_INCREMENT PRIMARY KEY,
  entity_type ENUM('user','ip') NOT NULL,
  entity      VARCHAR(128) NOT NULL,
  ts          DATETIME(3)  NOT NULL,
  score       DOUBLE       NOT NULL,               -- 특징별 |z| 의 최댓값
  features    JSON         NOT NULL,               -- 특징별 z 값
  KEY ix_anomaly_entity_ts (entity_type, entity, ts)
) ENGINE=InnoDB;

-- 6) 판정 이력 — 누가(LLM·캐시·규칙) 어떤 관문에서 판정했는가
CREATE TABLE IF NOT EXISTS triage_results (
  id                        BIGINT UNSIGNED AUTO_INCREMENT PRIMARY KEY,
  incident_id               BIGINT UNSIGNED NOT NULL,
  source                    ENUM('llm','cache','rule','refusal') NOT NULL,
  gate                      VARCHAR(16)  NOT NULL,   -- G1, G2, G3, G5, LLM (호출률·절감 측정용)
  model                     VARCHAR(64)  NULL,
  priority                  ENUM('P1','P2','P3','P4') NOT NULL,
  verdict                   ENUM('true_positive','benign','needs_review') NOT NULL,
  false_positive_likelihood DECIMAL(4,3) NULL,
  reasoning_ko              VARCHAR(600) NULL,
  recommended_playbook      ENUM('block_ip','lock_account','monitor','none') NOT NULL DEFAULT 'none',
  confidence                DECIMAL(4,3) NULL,
  created_at                DATETIME(3)  NOT NULL DEFAULT CURRENT_TIMESTAMP(3),
  KEY ix_triage_incident (incident_id, created_at),
  CONSTRAINT fk_triage_incident FOREIGN KEY (incident_id) REFERENCES incidents(id) ON DELETE CASCADE
) ENGINE=InnoDB;

-- 7) 지문 판정 캐시 (호출 깔때기 G3)
CREATE TABLE IF NOT EXISTS verdict_cache (
  fingerprint CHAR(64)     PRIMARY KEY,
  verdict     JSON         NOT NULL,               -- 판정 스키마 그대로
  model       VARCHAR(64)  NOT NULL,
  hits        INT UNSIGNED NOT NULL DEFAULT 0,
  created_at  DATETIME(3)  NOT NULL DEFAULT CURRENT_TIMESTAMP(3),
  expires_at  DATETIME(3)  NOT NULL,
  KEY ix_cache_expires (expires_at)
) ENGINE=InnoDB;

-- 8) LLM 호출 기록 — 토큰·캐시·비용·지연
CREATE TABLE IF NOT EXISTS llm_calls (
  id                 BIGINT UNSIGNED AUTO_INCREMENT PRIMARY KEY,
  ts                 DATETIME(3)   NOT NULL DEFAULT CURRENT_TIMESTAMP(3),
  provider           ENUM('claude','gemini') NOT NULL,
  model              VARCHAR(64)   NOT NULL,
  incidents_in_batch TINYINT UNSIGNED NOT NULL DEFAULT 1,
  input_tokens       INT UNSIGNED  NOT NULL DEFAULT 0,
  output_tokens      INT UNSIGNED  NOT NULL DEFAULT 0,
  cache_read_tokens  INT UNSIGNED  NOT NULL DEFAULT 0,
  cache_write_tokens INT UNSIGNED  NOT NULL DEFAULT 0,
  cost_usd_est       DECIMAL(10,6) NOT NULL DEFAULT 0,
  latency_ms         INT UNSIGNED  NULL,
  status             ENUM('ok','refusal','invalid','error','max_tokens') NOT NULL,
  gate_reason        VARCHAR(64)   NOT NULL,       -- 호출 사유(gray_zone, escalate_low_confidence ...)
  KEY ix_llm_ts (ts)
) ENGINE=InnoDB;

-- 9) 수집 체크포인트 — 중복·누락 없이 이어 받기
CREATE TABLE IF NOT EXISTS checkpoints (
  source     VARCHAR(32) PRIMARY KEY,              -- graylog:wazuh, graylog:board ...
  last_ts    DATETIME(3) NOT NULL,
  last_id    VARCHAR(64) NULL,
  updated_at DATETIME(3) NOT NULL DEFAULT CURRENT_TIMESTAMP(3) ON UPDATE CURRENT_TIMESTAMP(3)
) ENGINE=InnoDB;

-- 10) 대응 요청·승인·결과
CREATE TABLE IF NOT EXISTS actions (
  id               BIGINT UNSIGNED AUTO_INCREMENT PRIMARY KEY,
  incident_id      BIGINT UNSIGNED NOT NULL,
  playbook         ENUM('block_ip','lock_account','monitor') NOT NULL,
  requested_at     DATETIME(3)  NOT NULL DEFAULT CURRENT_TIMESTAMP(3),
  n8n_execution_id VARCHAR(64)  NULL,
  decision         ENUM('pending','approved','rejected','timeout') NOT NULL DEFAULT 'pending',
  decided_by       VARCHAR(128) NULL,
  decided_at       DATETIME(3)  NULL,
  executed_at      DATETIME(3)  NULL,
  changed          BOOLEAN      NULL,              -- 게시판 API 응답의 changed(멱등 200 함정 대비)
  result_note      VARCHAR(255) NULL,
  KEY ix_actions_incident (incident_id),
  CONSTRAINT fk_actions_incident FOREIGN KEY (incident_id) REFERENCES incidents(id) ON DELETE CASCADE
) ENGINE=InnoDB;
```
